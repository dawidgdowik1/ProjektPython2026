class Produkt:
    def __init__(self, nazwa, cena):
        self.nazwa = nazwa
        self.cena = cena

    @property
    def cena(self):
        return self._cena

    @cena.setter
    def cena(self, wartosc):
        if wartosc < 0:
            print(f"[Ostrzeżenie] Cena produktu '{self.nazwa}' nie może być ujemna! Ustawiono 0.0 zł.")
            self._cena = 0.0
        else:
            self._cena = float(wartosc)

    def pokaz_szczegoly(self):
        return f"Produkt: {self.nazwa} | Cena: {self.cena:.2f} zł"


if __name__ == "__main__":
    print("--- ROZPOCZĘCIE SYSTEMU ---")
    
    produkt1 = Produkt("Zwykła koszulka", 49.99)
    print(produkt1.pokaz_szczegoly())
    
    produkt2 = Produkt("Dziurawy kubek", -15.50)
    print(produkt2.pokaz_szczegoly())
